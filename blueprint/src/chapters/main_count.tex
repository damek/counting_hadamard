\chapter{Main theorem}
\label{chap:main-count}

Combining the primary estimate and the residual bound yields the main
asymptotic expansion.

\begin{theorem}[Main asymptotic count]\label{thm:main-intro}
\lean{thm_main_intro}\leanok
\uses{prop:primary-box,prop:residual-estimate,fact:fixed-n}
There exist $C_0,c>0$ such that for all sufficiently large $n$ and
$t\ge C_0 n^3$,
\[
  \frac{N_{n,4t}}{A_{n,4t}}
  =1-\frac{\binom{n}{3}}{8t}
  +O\!\left(\frac{n^2}{t}+\frac{n^{5/2}}{t^{3/2}}
  +\frac{n^6}{t^2}+e^{-cn^2}\right).
\]
\end{theorem}

\begin{proof}\leanok
\uses{prop:primary-box, prop:residual-estimate}
By the three-way decomposition,
$\Prb(S_{4t}=0)
=K_n\RePart\!\int_{\corereg}\psi^{4t}\,d\lambda+\RePart E$,
where $E$ collects off-core and residual contributions.
Proposition~\ref{prop:primary-box} gives the primary estimate.
Proposition~\ref{prop:residual-estimate} gives
$|E|=O(e^{-cn^2})\Ahat$, absorbed by the existing
$O(e^{-cn^2})$ term.
Since $N_{n,4t}=2^{4nt}\Prb(S_{4t}=0)$ and
$A_{n,4t}=2^{4nt}\Ahat$, dividing by $\Ahat$ yields the claim.
\end{proof}

\begin{corollary}[Uniform counting]\label{cor:uniform}
\lean{cor_uniform}\leanok
\uses{thm:main-intro,fact:fixed-n}
For every $\varepsilon>0$ there exists $K>0$ such that
\[
  \left|\frac{N_{n,4t}}{A_{n,4t}}-1\right|<\varepsilon
  \qquad\text{for all } n\ge 2 \text{ and } t\ge Kn^3.
\]
\end{corollary}

\begin{proof}\leanok
\uses{thm:main-intro,fact:fixed-n}
For $n\ge N_0$ (sufficiently large), the expansion in
Theorem~\ref{thm:main-intro} bounds the ratio by $\varepsilon$
when $K$ is large enough.
For each $n<N_0$, the fixed-$n$ asymptotics
(Lemma~\ref{fact:fixed-n}) provide a threshold $T_n$ beyond which
the ratio is within $\varepsilon$; enlarging $K$ so that
$Kn^3\ge T_n$ for all small $n$ completes the proof.
\end{proof}
