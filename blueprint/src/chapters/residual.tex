\chapter{Far shell and residual}\label{chap:residual}

This chapter bounds the contributions from outside the core region
$\corereg$. The far shell $\Bbox_{\pi/4}\setminus\Dd_r$ is split into
three sub-regions by the comparison error $J(\lambda)$ and the maximal
row influence $I_{\max}(\lambda)$.

\section{Far-shell contraction}

\begin{proposition}[Far-shell contraction]\label{prop:far-shell}
\lean{edgeEvenFarShell_contribution_bound_Jsplit}\leanok
\uses{cor:weak-comparison,fact:psi-sq,lem:gaussian-quadratic}
Set $\eta_r:=\frac{1}{3}[1-(1+2r^2)^{-1/4}]$ and
$I_{\max}(\lambda):=\max_k I_k(\lambda)$.
There exist $q_{\mathrm{sm}},q_{\mathrm{big}}\in(0,1)$ and $a_r>0$
such that for all $n\ge 2$ and $t\ge 1$ the far shell decomposes as
\[
  \begin{aligned}
  \mathcal F_{\mathrm{sm}}
    &:=\{\lambda\in\Bbox_{\pi/4}\setminus\Dd_r:J(\lambda)\le \eta_r\},\\
  \mathcal F_{\mathrm{big}}
    &:=\{\lambda\in\Bbox_{\pi/4}\setminus\Dd_r:J(\lambda)>\eta_r,\;I_{\max}\ge 1\},\\
  \mathcal F_{\mathrm{mid}}
    &:=\{\lambda\in\Bbox_{\pi/4}\setminus\Dd_r:J(\lambda)>\eta_r,\;I_{\max}<1\},
  \end{aligned}
\]
and
\[
  |\psi(\lambda)|^{4t}\le
  \begin{cases}
    q_{\mathrm{sm}}^{4t} & \lambda\in\mathcal F_{\mathrm{sm}},\\
    q_{\mathrm{big}}^{4t} & \lambda\in\mathcal F_{\mathrm{big}},\\
    \exp(-a_r t n^{-2/3}) & \lambda\in\mathcal F_{\mathrm{mid}}.
  \end{cases}
\]
\end{proposition}

\begin{proof}\leanok
\uses{cor:weak-comparison,fact:psi-sq,lem:gaussian-quadratic}
\emph{Small-$J$:}
Lemma~\ref{lem:gaussian-quadratic} gives
$|\psi_G(\lambda)|\le(1+2s(\lambda))^{-1/4}\le q_G$.
Corollary~\ref{cor:weak-comparison} gives
$|\psi-\psi_G|\le\frac32 J\le\frac32\eta_r=(1-q_G)/2$,
so $|\psi|\le q_{\mathrm{sm}}=(1+q_G)/2<1$.

\emph{Large-$I_{\max}$:}
Lemma~\ref{fact:psi-sq} gives
$|\psi|^2\le\frac12+\frac12 e^{-8I_{\max}/\pi^2}\le q_{\mathrm{big}}^2$.

\emph{Middle:}
The same cosine-product bound gives
$|\psi|^2\le e^{-2I_{\max}/\pi^2}$, and
$J>  \eta_r$ with $J\le nI_{\max}^{3/2}$ implies
$I_{\max}\ge(\eta_r/n)^{2/3}$, so
$|\psi|^{4t}\le\exp(-a_r tn^{-2/3})$.
\end{proof}

\section{Far-shell integrals}

\begin{corollary}[Far-shell integrals]\label{cor:far-negligible}
\lean{residual_estimate_quantitative}\leanok
\uses{prop:far-shell,fact:lambda-facts}
Let $E_r(n,t):=q_{\mathrm{sm}}^{4t}+q_{\mathrm{big}}^{4t}+e^{-a_r t n^{-2/3}}$.
Then for every $\delta\in(0,\pi/4)$,
\[
  \begin{aligned}
  K_n\int_{\Bbox_\delta\setminus\Dd_r}|\psi(\lambda)|^{4t}\,d\lambda
    &\le 2^{-n+1}E_r(n,t),\\
  (2\pi)^{-d}\sum_{\lambda\in\Lambda}
    \int_{\lambda+(\Bbox_{\pi/4}\setminus\Dd_r)}|\psi(\gamma)|^{4t}\,d\gamma
    &\le 2^{-n+1}E_r(n,t).
  \end{aligned}
\]
Both are $o(\Ahat)$ as $n\to\infty$ with $t/(n^{8/3}\log t)\to\infty$.
\end{corollary}

\begin{proof}\leanok
\uses{prop:far-shell,fact:lambda-facts}
Integrate the pointwise bounds from Proposition~\ref{prop:far-shell}
over $\Bbox_{\pi/4}\setminus\Dd_r$ (volume $\le(\pi/2)^d$).
For the translated shell, the multiplicativity
$\psi(\lambda+\mu)=\psi(\lambda)\psi(\mu)$ and $|\psi(\lambda)|=1$ for
$\lambda\in\Lambda$ reduce to the same integral.
The ratio $2^{-n+1}E_r/\Ahat\to 0$ since
$t/(n^{8/3}\log t)\to\infty$ ensures
$tn^{-2/3}\gg d\log t$.
\end{proof}

\section{Residual estimate}

\begin{proposition}[Residual estimate]\label{prop:residual-estimate}
\lean{residual_estimate_quantitative}\leanok
\uses{lem:small-ball,cor:far-negligible,lem:odd-bound,lem:inner-core}
Set $\delta^2=2d/t$ and $K_n:=2^{2d-n+1}(2\pi)^{-d}$. Then
\[
  \begin{aligned}
    K_n\int_{\Bbox_\delta\setminus\corereg}|\psi(\lambda)|^{4t}\,d\lambda
    &= o(\Ahat),\\
    (2\pi)^{-d}\left|\int_{R_\delta}\psi(\gamma)^{4t}\,d\gamma\right|
    &= o(\Ahat),
  \end{aligned}
\]
as $n\to\infty$ with $t/(n^{8/3}\log t)\to\infty$.
For $t\ge C_0 n^3$ and all sufficiently large $n$, both bounds
are at most $Ce^{-cn^2}\Ahat$.
\end{proposition}

\begin{proof}\leanok
\uses{lem:small-ball,cor:far-negligible,lem:odd-bound}
\emph{Outside the core:}
If $\lambda\in(\Bbox_\delta\cap\Dd_r)\setminus\corereg$, then
$\|\lambda\|^2>d/t$, so Lemma~\ref{lem:small-ball} gives
$|\psi|^{4t}\le e^{-d/2}e^{-t\|\lambda\|^2}$, and the integral is
$O(e^{-cd})\core(d,t)$.

\emph{Far shell:}
Corollary~\ref{cor:far-negligible} bounds the contribution of
$\Bbox_\delta\setminus\Dd_r$.

\emph{Residual $R_\delta$:}
Decompose into even cells (handled by small-ball decay + far-shell
bounds) and odd cells (Lemma~\ref{lem:odd-bound} gives
$|\psi|^{4t}\le 2^{-2t}$).

At the cubic threshold $t\ge C_0 n^3$, all bounds are $O(e^{-cn^2})\Ahat$.
\end{proof}
